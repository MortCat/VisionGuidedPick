cmake_minimum_required(VERSION 3.16)
project(factory_gripper_adapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(control_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
add_executable(gripper_adapter gripper_adapter_main.cpp)
ament_target_dependencies(gripper_adapter rclcpp rclcpp_action control_msgs sensor_msgs trajectory_msgs)
install(TARGETS gripper_adapter DESTINATION lib/${PROJECT_NAME})
ament_package()
