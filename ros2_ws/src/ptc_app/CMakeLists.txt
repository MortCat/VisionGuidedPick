cmake_minimum_required(VERSION 3.16)
project(ptc_app)
include(cmake/ptc_dependencies.cmake)

if(NOT PTC_STANDALONE)
  find_package(ament_cmake REQUIRED)
  find_package(ptc_core REQUIRED)
endif()
find_package(yaml-cpp CONFIG REQUIRED)
find_package(OpenSSL 3 REQUIRED)
add_library(ptc_settings STATIC src/settings_loader.cpp src/settings_check_runner.cpp src/scene_loader.cpp src/reference_cloud_loader.cpp)
target_compile_features(ptc_settings PUBLIC cxx_std_17)
target_include_directories(ptc_settings PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
if(PTC_STANDALONE)
  target_link_libraries(ptc_settings PUBLIC ptc_core)
else()
  target_link_libraries(ptc_settings PUBLIC ptc_core::ptc_core)
endif()
target_link_libraries(ptc_settings PRIVATE yaml-cpp::yaml-cpp OpenSSL::Crypto)
if(MSVC)
  target_compile_options(ptc_settings PRIVATE /W4 /permissive-)
else()
  target_compile_options(ptc_settings PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(settings_check src/settings_check_main.cpp)
target_link_libraries(settings_check PRIVATE ptc_settings)
install(TARGETS ptc_settings EXPORT export_ptc_settings ARCHIVE DESTINATION lib)
install(TARGETS settings_check RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  include(CTest)
  add_executable(settings_loader_test test/settings_loader_test.cpp)
  target_include_directories(settings_loader_test PRIVATE ../ptc_core/test)
  target_link_libraries(settings_loader_test PRIVATE ptc_settings yaml-cpp::yaml-cpp)
  add_test(NAME settings_loader_test COMMAND settings_loader_test
    ${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures ${CMAKE_CURRENT_BINARY_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../../../config/settings.yaml)
  add_test(NAME settings_draft_rejected COMMAND settings_check ${CMAKE_CURRENT_SOURCE_DIR}/../../../config/settings.yaml)
  set_tests_properties(settings_draft_rejected PROPERTIES WILL_FAIL TRUE)
endif()

if(NOT PTC_STANDALONE)
  ament_export_targets(export_ptc_settings HAS_LIBRARY_TARGET)
  find_package(ptc_interfaces REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(urdf REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(PNG REQUIRED)
  add_library(ptc_ros STATIC src/ros_converter.cpp src/robot_description_loader.cpp src/node_runtime.cpp src/rgbd_converter.cpp)
  target_compile_features(ptc_ros PUBLIC cxx_std_17)
  target_include_directories(ptc_ros PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  target_link_libraries(ptc_ros PUBLIC ptc_settings yaml-cpp::yaml-cpp PNG::PNG)
  ament_target_dependencies(ptc_ros PUBLIC ptc_interfaces sensor_msgs urdf rclcpp)
  install(TARGETS ptc_ros EXPORT export_ptc_ros ARCHIVE DESTINATION lib)
  if(BUILD_TESTING)
    add_executable(robot_description_loader_test test/robot_description_loader_test.cpp)
    target_link_libraries(robot_description_loader_test PRIVATE ptc_ros)
    ament_target_dependencies(robot_description_loader_test PUBLIC urdf)
    add_test(NAME robot_description_loader_test COMMAND robot_description_loader_test
      ${CMAKE_CURRENT_SOURCE_DIR}/../../../config/factory_demo/arm_01.yaml)
  endif()
  ament_export_targets(export_ptc_ros HAS_LIBRARY_TARGET)
  ament_export_dependencies(ptc_interfaces sensor_msgs urdf rclcpp PNG)
  ament_export_dependencies(ptc_core yaml-cpp OpenSSL)
  ament_package(CONFIG_EXTRAS cmake/ptc_dependencies.cmake)
endif()
