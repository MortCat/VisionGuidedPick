cmake_minimum_required(VERSION 3.16)
project(ptc_motion)
find_package(ament_cmake REQUIRED)
find_package(ptc_app REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_library(ptc_motion STATIC src/arm_gateway_node.cpp)
target_compile_features(ptc_motion PUBLIC cxx_std_17)
target_include_directories(ptc_motion PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(ptc_motion PUBLIC ptc_app::ptc_ros)
ament_target_dependencies(ptc_motion PUBLIC rclcpp_action tf2_ros tf2_geometry_msgs)
install(TARGETS ptc_motion EXPORT export_ptc_motion ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
ament_export_targets(export_ptc_motion HAS_LIBRARY_TARGET)
ament_export_dependencies(ptc_app rclcpp_action tf2_ros tf2_geometry_msgs)
ament_package()
