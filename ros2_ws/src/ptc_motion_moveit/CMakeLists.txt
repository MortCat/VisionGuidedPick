cmake_minimum_required(VERSION 3.16)
project(ptc_motion_moveit)
find_package(ament_cmake REQUIRED)
find_package(ptc_motion REQUIRED)
find_package(moveit_ros_planning_interface REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(control_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(geometric_shapes REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_executable(arm_gateway_moveit src/arm_gateway_moveit_main.cpp src/moveit_motion_backend.cpp)
target_compile_features(arm_gateway_moveit PRIVATE cxx_std_17)
target_include_directories(arm_gateway_moveit PRIVATE include)
target_link_libraries(arm_gateway_moveit PRIVATE ptc_motion::ptc_motion)
ament_target_dependencies(arm_gateway_moveit PUBLIC moveit_ros_planning_interface moveit_msgs control_msgs std_srvs geometric_shapes tf2_geometry_msgs)
install(TARGETS arm_gateway_moveit DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(grasp_clearance_model_test test/grasp_clearance_model_test.cpp)
  target_compile_features(grasp_clearance_model_test PRIVATE cxx_std_17)
  target_include_directories(grasp_clearance_model_test PRIVATE include)
  ament_target_dependencies(grasp_clearance_model_test geometric_shapes)
  add_test(NAME grasp_clearance_model_test COMMAND grasp_clearance_model_test)
  add_executable(grasp_contact_policy_test test/grasp_contact_policy_test.cpp)
  target_compile_features(grasp_contact_policy_test PRIVATE cxx_std_17)
  target_include_directories(grasp_contact_policy_test PRIVATE include)
  ament_target_dependencies(grasp_contact_policy_test moveit_msgs)
  add_test(NAME grasp_contact_policy_test COMMAND grasp_contact_policy_test)
  add_executable(gripper_feedback_policy_test test/gripper_feedback_policy_test.cpp)
  target_compile_features(gripper_feedback_policy_test PRIVATE cxx_std_17)
  target_include_directories(gripper_feedback_policy_test PRIVATE include)
  add_test(NAME gripper_feedback_policy_test COMMAND gripper_feedback_policy_test)
endif()
ament_package()
