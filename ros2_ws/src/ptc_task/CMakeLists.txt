cmake_minimum_required(VERSION 3.16)
project(ptc_task)
find_package(ament_cmake REQUIRED)
find_package(ptc_app REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(task_coordinator src/task_coordinator_main.cpp src/task_coordinator_node.cpp)
target_compile_features(task_coordinator PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(task_coordinator PRIVATE /bigobj)
endif()
target_include_directories(task_coordinator PRIVATE include)
target_link_libraries(task_coordinator PRIVATE ptc_app::ptc_ros)
ament_target_dependencies(task_coordinator PUBLIC rclcpp_action tf2_ros)
install(TARGETS task_coordinator DESTINATION lib/${PROJECT_NAME})
ament_package()
