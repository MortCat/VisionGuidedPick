cmake_minimum_required(VERSION 3.16)
project(ptc_vision)
find_package(ament_cmake REQUIRED)
find_package(ptc_app REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_executable(vision src/vision_main.cpp src/vision_node.cpp src/cloud_registration_adapter.cpp)
target_compile_features(vision PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(vision PRIVATE /bigobj)
endif()
target_include_directories(vision PRIVATE include)
target_link_libraries(vision PRIVATE ptc_app::ptc_ros)
ament_target_dependencies(vision PUBLIC rclcpp_action tf2_ros tf2_geometry_msgs)
install(TARGETS vision DESTINATION lib/${PROJECT_NAME})
find_package(ptc_core REQUIRED)
find_package(jsoncpp REQUIRED)
add_executable(observe_geometry src/observe_geometry_main.cpp)
target_compile_features(observe_geometry PRIVATE cxx_std_17)
target_link_libraries(observe_geometry PRIVATE ptc_core::ptc_core JsonCpp::JsonCpp)
install(TARGETS observe_geometry DESTINATION lib/${PROJECT_NAME})
ament_package()
